    const t=setInterval(()=>{
      setTime(v=>{
        if(v<=1){
          const win=gameInfo.items[Math.floor(Math.random()*gameInfo.items.length)];

          setBets({}); return 10;
        } return v-1;
      })

    if(time<=2) return alert('BET TIME KHATAM! Agle round ka wait karo');
